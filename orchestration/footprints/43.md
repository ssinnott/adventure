  :59-71 is not touched) and `meet(p, party): string`. It depends on the person and the party alone
  (no Game, World, rng or screens), changes the party as `takeItem` does and returns the words.
  It imports only `./party.ts` (`countItem` :295, `takeItem` :302) and types from `./map.ts`. The
  precedence is today's (after, then the hand-in, then the lines that hire); only the needFlag test
  goes, and "early" is the person's own hire `flag` still unset:

  ```ts
  export function meet(p: Person, party: Party): string {
    const q = p.quest;
    if (q && party.flags[q.setFlag]) return q.after.join('\n\n');
    if (q && countItem(party, q.item) > 0) {
      const early = !!p.flag && !party.flags[p.flag];
      takeItem(party, q.item); party.gold += q.reward; party.flags[q.setFlag] = 1;
      return (early && q.early ? q.early : q.done).join('\n\n') + `\n\n(${q.reward} gold.)`;
    }
    if (p.flag && !party.flags[p.flag]) party.flags[p.flag] = 1;
    return p.lines.join('\n\n');
  }
  ```
